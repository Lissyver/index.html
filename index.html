// ===================================================== // CADA DÍA CON DIOS // MINI APP → GOOGLE APPS SCRIPT → GOOGLE SHEETS // ===================================================== // ⚠️ COLOCA AQUÍ EL TOKEN NUEVO DE BOTFATHER const BOT_TOKEN = 'TU_TOKEN_NUEVO'; const SHEET_ID = '1KMlDV8Tf67KlaQ6kQalegWY4UPMeVJ815T0dlTiim-M'; const ZONA_HORARIA = 'America/Panama'; // ===================================================== // DO GET // ===================================================== function doGet(e) { return ContentService .createTextOutput( JSON.stringify({ ok: true, servicio: 'Cada Día con Dios', estado: 'activo' }) ) .setMimeType(ContentService.MimeType.JSON); } // ===================================================== // DO POST // RECIBE INFORMACIÓN DE LA MINI APP // ===================================================== function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return respuestaJSON({ ok: false, error: 'No se recibió información' }); } const datos = JSON.parse(e.postData.contents); const tipo = datos.tipo || ''; const detalle = datos.detalle || ''; const telegramId = datos.telegramId || ''; const nombre = datos.nombre || 'Peregrino'; const username = datos.username || ''; // ------------------------------------------------- // REGISTRAR USUARIO // ------------------------------------------------- if (telegramId) { registrarUsuario( telegramId, nombre, username ); } // ------------------------------------------------- // REGISTRO DIARIO // ------------------------------------------------- if (tipo === 'Registro Diario') { const resultado = registrarDia( telegramId, nombre ); return respuestaJSON(resultado); } // ------------------------------------------------- // OTROS EVENTOS // ------------------------------------------------- if (tipo) { registrarEvento( telegramId, nombre, tipo, detalle ); return respuestaJSON({ ok: true, evento: tipo }); } return respuestaJSON({ ok: true }); } catch (error) { console.error(error); return respuestaJSON({ ok: false, error: error.toString() }); } } // ===================================================== // RESPUESTA JSON // ===================================================== function respuestaJSON(objeto) { return ContentService .createTextOutput( JSON.stringify(objeto) ) .setMimeType(ContentService.MimeType.JSON); } // ===================================================== // REGISTRAR USUARIO // ===================================================== function registrarUsuario( telegramId, nombre, username ) { const ss = SpreadsheetApp.openById(SHEET_ID); let hoja = ss.getSheetByName('Usuarios'); if (!hoja) { hoja = ss.insertSheet('Usuarios'); hoja.appendRow([ 'Chat ID', 'Nombre', 'Username', 'Fecha de Registro' ]); } const datos = hoja.getDataRange().getValues(); for (let i = 1; i < datos.length; i++) { if ( String(datos[i][0]) === String(telegramId) ) { return; } } hoja.appendRow([ telegramId, nombre, username, new Date() ]); } // ===================================================== // REGISTRAR DÍA // ===================================================== function registrarDia( telegramId, nombre ) { if (!telegramId) { return { ok: false, error: 'No se recibió Telegram ID' }; } const ss = SpreadsheetApp.openById(SHEET_ID); let hoja = ss.getSheetByName('Registros'); if (!hoja) { hoja = ss.insertSheet('Registros'); hoja.appendRow([ 'Fecha', 'Chat ID', 'Nombre', 'Tipo de Evento', 'Día de Racha' ]); } const hoy = Utilities.formatDate( new Date(), ZONA_HORARIA, 'yyyy-MM-dd' ); const datos = hoja.getDataRange().getValues(); // ------------------------------------------------- // COMPROBAR SI YA REGISTRÓ HOY // ------------------------------------------------- for (let i = 1; i < datos.length; i++) { const fechaCelda = datos[i][0]; const id = String(datos[i][1]); if (!fechaCelda) { continue; } const fecha = Utilities.formatDate( new Date(fechaCelda), ZONA_HORARIA, 'yyyy-MM-dd' ); if ( fecha === hoy && id === String(telegramId) ) { const rachaActual = datos[i][4] || 1; return { ok: true, yaRegistrado: true, racha: Number(rachaActual), mensaje: 'Ya registraste tu oración de hoy.' }; } } // ------------------------------------------------- // CALCULAR RACHA // ------------------------------------------------- const racha = calcularRacha( hoja, telegramId, hoy ); // ------------------------------------------------- // GUARDAR REGISTRO // ------------------------------------------------- hoja.appendRow([ new Date(), telegramId, nombre, 'Registro diario', racha ]); // ------------------------------------------------- // DETERMINAR PREMIO // ------------------------------------------------- let premio = ''; if (racha === 7) { premio = '7 días'; } if (racha === 14) { premio = '14 días'; } if (racha === 21) { premio = '21 días'; } if (racha === 40) { premio = '40 días'; } // ------------------------------------------------- // ENVIAR MENSAJE A TELEGRAM // ------------------------------------------------- let mensaje = '🙏 *¡Registro completado!*\n\n' + 'Hola, ' + nombre + '. ❤️\n\n' + '🔥 Tu racha actual es de *' + racha + ' día(s)*.\n\n'; if (premio) { mensaje += '🎁 *¡FELICIDADES!*\n\n' + 'Has completado *' + premio + '* de oración.\n\n' + 'Tu recompensa ha sido desbloqueada. 🎉\n\n'; } mensaje += 'Sigue caminando cada día con Dios. 🙏'; enviarMensaje( telegramId, mensaje ); return { ok: true, yaRegistrado: false, racha: racha, premio: premio, mensaje: 'Registro completado' }; } // ===================================================== // CALCULAR RACHA // ===================================================== function calcularRacha( hoja, telegramId, hoy ) { const datos = hoja.getDataRange().getValues(); const fechas = []; for (let i = 1; i < datos.length; i++) { const id = String(datos[i][1]); if ( id !== String(telegramId) ) { continue; } if (!datos[i][0]) { continue; } const fecha = Utilities.formatDate( new Date(datos[i][0]), ZONA_HORARIA, 'yyyy-MM-dd' ); if (!fechas.includes(fecha)) { fechas.push(fecha); } } fechas.sort().reverse(); // Si no hay registros anteriores if (fechas.length === 0) { return 1; } let racha = 1; let fechaAnterior = new Date(hoy + 'T12:00:00'); for ( let i = 0; i < fechas.length; i++ ) { const fechaRegistro = fechas[i]; // El registro de hoy no se cuenta if ( fechaRegistro === hoy ) { continue; } const fecha = new Date( fechaRegistro + 'T12:00:00' ); const diferencia = Math.round( ( fechaAnterior - fecha ) / (1000 * 60 * 60 * 24) ); if (diferencia === 1) { racha++; fechaAnterior = fecha; } else { break; } } return racha; } // ===================================================== // REGISTRAR OTROS EVENTOS // ===================================================== function registrarEvento( telegramId, nombre, tipo, detalle ) { const ss = SpreadsheetApp.openById(SHEET_ID); let hoja = ss.getSheetByName('Eventos'); if (!hoja) { hoja = ss.insertSheet('Eventos'); hoja.appendRow([ 'Fecha', 'Chat ID', 'Nombre', 'Tipo', 'Detalle' ]); } hoja.appendRow([ new Date(), telegramId, nombre, tipo, detalle ]); } // ===================================================== // ENVIAR MENSAJE A TELEGRAM // ===================================================== function enviarMensaje( chatId, texto ) { if (!chatId) { return; } const url = 'https://api.telegram.org/bot' + BOT_TOKEN + '/sendMessage'; const payload = { chat_id: chatId, text: texto, parse_mode: 'Markdown' }; UrlFetchApp.fetch( url, { method: 'post', contentType: 'application/json', payload: JSON.stringify(payload), muteHttpExceptions: true } ); } // ===================================================== // RECORDATORIO DIARIO // ===================================================== function enviarRecordatorioDiario() { const ss = SpreadsheetApp.openById(SHEET_ID); const hoja = ss.getSheetByName('Usuarios'); if (!hoja) { return; } const datos = hoja.getDataRange().getValues(); const mensaje = '🙏 ¡Hola, peregrino!\n\n' + 'Este es tu recordatorio de *Cada Día con Dios*.\n\n' + 'Entra a la Mini App y aparta hoy tu momento de oración. 🔥'; for ( let i = 1; i < datos.length; i++ ) { const chatId = datos[i][0]; if (!chatId) { continue; } enviarMensaje( chatId, mensaje ); } }
